package org.ddolib.common.solver.stat;

import org.ddolib.common.util.PrettyPrint;

import java.text.DecimalFormat;
import java.text.DecimalFormatSymbols;
import java.util.List;
import java.util.Locale;

import static java.lang.Math.abs;

/**
 * Class representing the statistics of a search process in a solver.
 *
 * <p>This class tracks various metrics such as runtime, number of iterations, incumbent value, and
 * optimality gap. It is designed to be immutable-like, where update methods return a new instance
 * with the updated value.
 *
 * <p>It represents the search statistics of a solver. Subclasses override the update methods to
 * return their specific type via covariant return types.
 */
@SuppressWarnings("checkstyle:MemberName") // protected fields of the public API (_startTime...)
public abstract class SearchStatistics {
    /**
     * Start time of the search (in milliseconds).
     */
    protected final long _startTime;

    /**
     * Time at which the last improvement was found (in milliseconds).
     */
    protected long _lastTimeOfImprovement;

    /**
     * Current time of the search (in milliseconds).
     */
    protected long _currentTime;

    /**
     * Current status of the search.
     */
    protected SearchStatus _status = SearchStatus.UNKNOWN;

    /**
     * Total number of iterations performed.
     */
    protected int _nbIterations = 0;

    /**
     * Iteration during which the last improvement was found.
     */
    protected int _lastIterationOfImprovement = 0;

    /**
     * Value of the best solution found so far (incumbent).
     */
    protected double _incumbent;

    /**
     * Value of the previous incumbent found.
     */
    protected double _prevIncumbent = Double.POSITIVE_INFINITY;

    /**
     * Best optimality gap reached so far.
     */
    protected double _gap = Double.POSITIVE_INFINITY;

    /**
     * Best lower bound on the optimal value proved so far (the problems are minimized).
     */
    protected double _bestBound = Double.NEGATIVE_INFINITY;

    /**
     * Number of nodes expanded so far (nodes whose successors were computed).
     */
    protected long _nbExpandedNodes = 0;

    /**
     * Number of nodes generated so far (successors computed, before any pruning).
     */
    protected long _nbGeneratedNodes = 0;

    /**
     * Maximum size reached by the search frontier.
     */
    protected int _frontierMaxSize = 0;

    /**
     * Relative improvement of the incumbent value.
     */
    protected double _relativeImprovement = 1.0;

    /**
     * Iteration during which the last gap improvement was found.
     */
    protected int _lastIterationOfGapImprovement = 0;

    /**
     * Number of nodes pruned because their fast lower bound shows that they cannot improve the
     * best solution found so far.
     */
    protected int _nbFastLowerBoundPruning = 0;

    /** Number of nodes pruned because their state is dominated by the state of another node. */
    protected int _nbDominancePruning = 0;

    /**
     * Constructs a new SearchStatistics instance.
     *
     * @param startTime the start time of the search (in milliseconds)
     * @param initValue the initial value for the incumbent
     */
    public SearchStatistics(long startTime, double initValue) {
        _startTime = startTime;
        _lastTimeOfImprovement = startTime;
        _currentTime = startTime;
        _incumbent = initValue;
    }

    /**
     * Creates a new instance of the specific subclass with initial values. This is used by the copy
     * method to maintain the correct type.
     *
     * @return a new instance of the specific subclass
     */
    protected abstract SearchStatistics createSpecificInstance();

    /**
     * Returns the time at which the last improvement was found.
     *
     * @return the last time of improvement (in milliseconds)
     */
    public long lastTimeOfImprovement() {
        return _lastTimeOfImprovement;
    }

    /**
     * Returns the total runtime of the search.
     *
     * @return the runtime (in milliseconds)
     */
    public long runtime() {
        return _currentTime - _startTime;
    }

    /**
     * Returns the current time recorded in these statistics.
     *
     * @return the current time (in milliseconds)
     */
    public long currentTime() {
        return _currentTime;
    }

    /**
     * Returns the total number of iterations performed by the search.
     *
     * @return the number of iterations
     */
    public int nbIterations() {
        return _nbIterations;
    }

    /**
     * Returns the iteration number during which the last improvement was found.
     *
     * @return the last iteration of improvement
     */
    public int lastIterationOfImprovement() {
        return _lastIterationOfImprovement;
    }

    /**
     * Returns the iteration number during which the last gap improvement was found.
     *
     * @return the last iteration of gap improvement
     */
    public int lastIterationOfGapImprovement() {
        return _lastIterationOfGapImprovement;
    }

    /**
     * Returns the current optimality gap.
     *
     * @return the gap (percentage)
     */
    public double gap() {
        return _gap;
    }

    /**
     * Returns the number of nodes expanded so far: the nodes whose successors were computed. For
     * A*, AWA*, ACS and DFS, a node removed from the open list but then discarded (dominated,
     * already closed or pruned by the bound) is not expanded. For the DD-based solvers (DDO, LNS,
     * exact), it is the number of DD nodes whose children were generated, over all the compiled
     * DDs (restricted and relaxed).
     *
     * <p>With {@link #nbGeneratedNodes()}, it allows to compare the search effort of different
     * solvers, for instance with the {@code expanded} and {@code generated} counts of DIDP.
     *
     * @return the number of expanded nodes
     */
    public long nbExpandedNodes() {
        return _nbExpandedNodes;
    }

    /**
     * Returns the number of nodes generated so far: one per successor computed when a node is
     * expanded (one per transition applied), before any pruning by the bound, the dominance or a
     * duplicate. For the DD-based solvers, a transition leading to an existing node of the next
     * layer is also counted. The root is not counted.
     *
     * @return the number of generated nodes
     */
    public long nbGeneratedNodes() {
        return _nbGeneratedNodes;
    }

    /**
     * Returns the best lower bound on the optimal value proved so far (all the problems are
     * minimized). It is known even when the search is stopped before the end, for instance by a
     * time limit, and does not require an incumbent.
     *
     * <ul>
     *   <li>When the search is {@link SearchStatus#OPTIMAL}, it is the incumbent value.
     *   <li>When the problem is {@link SearchStatus#UNSAT}, it is {@link Double#POSITIVE_INFINITY}.
     *   <li>When no bound is known, it is {@link Double#NEGATIVE_INFINITY}.
     * </ul>
     *
     * @return the best lower bound on the optimal value
     */
    public double bestBound() {
        if (_status == SearchStatus.OPTIMAL) {
            return _incumbent;
        } else if (_status == SearchStatus.UNSAT) {
            return Double.POSITIVE_INFINITY;
        }
        return _bestBound;
    }

    /**
     * Returns the value of the best solution found so far (incumbent).
     *
     * @return the incumbent value
     */
    public double incumbent() {
        return _incumbent;
    }

    /**
     * Returns the value of the previous incumbent found before the current one.
     *
     * @return the previous incumbent value
     */
    public double prevIncumbent() {
        return _prevIncumbent;
    }

    /**
     * Returns the current status of the search.
     *
     * @return the {@link SearchStatus}
     */
    public SearchStatus status() {
        return _status;
    }

    /**
     * Returns the maximum size reached by the search frontier during the process.
     *
     * @return the maximum frontier size
     */
    public int frontierMaxSize() {
        return _frontierMaxSize;
    }

    /**
     * Returns the relative improvement of the last incumbent update in percent.
     *
     * @return the relative improvement in percent
     */
    public double relativeImprovement() {
        return 100 * _relativeImprovement;
    }

    /**
     * Returns the number of nodes pruned because their fast lower bound shows that they cannot
     * improve the best solution found so far.
     *
     * @return the number of nodes pruned by the fast lower bound
     */
    public int nbFastLowerBoundPruning() {
        return _nbFastLowerBoundPruning;
    }

    /**
     * Returns the number of nodes pruned because their state is dominated by the state of another
     * node.
     *
     * @return the number of nodes pruned by the dominance
     */
    public int nbDominancePruning() {
        return _nbDominancePruning;
    }

    /**
     * Creates and returns a copy of this statistics instance.
     *
     * @return a copy of the current statistics
     */
    public SearchStatistics copy() {
        SearchStatistics clone = createSpecificInstance();

        clone._currentTime = this._currentTime;
        clone._lastTimeOfImprovement = this._lastTimeOfImprovement;
        clone._status = this._status;
        clone._nbIterations = this._nbIterations;
        clone._lastIterationOfImprovement = this._lastIterationOfImprovement;
        clone._lastIterationOfGapImprovement = this._lastIterationOfGapImprovement;
        clone._prevIncumbent = this._prevIncumbent;
        clone._gap = this._gap;
        clone._bestBound = this._bestBound;
        clone._nbExpandedNodes = this._nbExpandedNodes;
        clone._nbGeneratedNodes = this._nbGeneratedNodes;
        clone._frontierMaxSize = this._frontierMaxSize;
        clone._relativeImprovement = this._relativeImprovement;
        clone._nbFastLowerBoundPruning = this._nbFastLowerBoundPruning;
        clone._nbDominancePruning = this._nbDominancePruning;

        return clone;
    }

    /**
     * Returns a new SearchStatistics instance with an updated incumbent value and gap.
     *
     * @param incumbent the new incumbent value
     * @param gap       the new optimality gap
     * @return a new instance with updated incumbent and gap
     */
    public SearchStatistics updateIncumbent(double incumbent, double gap) {
        SearchStatistics toReturn = this.copy();
        toReturn._incumbent = incumbent;
        toReturn._prevIncumbent = this._incumbent;
        toReturn._currentTime = System.currentTimeMillis();
        toReturn._lastTimeOfImprovement = toReturn._currentTime;
        toReturn._lastIterationOfImprovement = this._nbIterations;
        if (gap < this._gap) {
            toReturn._lastIterationOfGapImprovement = this._nbIterations;
            toReturn._gap = gap;
        }
        if (Double.isInfinite(this._incumbent) || this._incumbent == 0) {
            toReturn._relativeImprovement = 1.0;
        } else {
            toReturn._relativeImprovement = abs(incumbent - this._incumbent) / abs(this._incumbent);
        }

        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with an updated search status.
     *
     * @param status the new {@link SearchStatus}
     * @return a new instance with the updated status
     */
    public SearchStatistics updateStatus(SearchStatus status) {
        SearchStatistics toReturn = this.copy();
        toReturn._status = status;
        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with the iteration count incremented by one.
     *
     * @return a new instance with incremented iterations
     */
    public SearchStatistics incrementNbIter() {
        SearchStatistics toReturn = this.copy();
        toReturn._nbIterations++;
        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with an updated maximum frontier size.
     *
     * @param frontierSize the current frontier size to potentially update the maximum
     * @return a new instance with the updated maximum frontier size
     */
    public SearchStatistics updateFrontierMaxSize(int frontierSize) {
        SearchStatistics toReturn = this.copy();
        toReturn._frontierMaxSize = Integer.max(this._frontierMaxSize, frontierSize);
        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with an updated optimality gap.
     *
     * @param gap the new gap value
     * @return a new instance with the updated gap
     */
    public SearchStatistics updateGap(double gap) {
        SearchStatistics toReturn = this.copy();
        if (gap < this._gap) {
            toReturn._lastIterationOfGapImprovement = this._nbIterations;
            toReturn._gap = gap;
        }
        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with an updated best bound. Any valid lower bound on
     * the optimal value can be given: the best one seen so far is kept.
     *
     * @param bound a lower bound on the optimal value
     * @return a new instance with the updated best bound
     */
    public SearchStatistics updateBestBound(double bound) {
        SearchStatistics toReturn = this.copy();
        if (bound > this._bestBound) {
            toReturn._bestBound = bound;
        }
        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with the given numbers of expanded and generated
     * nodes (totals since the beginning of the search).
     *
     * @param nbExpanded the number of expanded nodes
     * @param nbGenerated the number of generated nodes
     * @return a new instance with the updated node counts
     */
    public SearchStatistics updateNodeCounts(long nbExpanded, long nbGenerated) {
        SearchStatistics toReturn = this.copy();
        toReturn._nbExpandedNodes = nbExpanded;
        toReturn._nbGeneratedNodes = nbGenerated;
        return toReturn;
    }

    /**
     * Returns a new SearchStatistics instance with the given numbers of expanded and generated
     * nodes added to the current ones.
     *
     * @param nbExpanded the number of expanded nodes to add
     * @param nbGenerated the number of generated nodes to add
     * @return a new instance with the updated node counts
     */
    public SearchStatistics addNodeCounts(long nbExpanded, long nbGenerated) {
        return updateNodeCounts(
                _nbExpandedNodes + nbExpanded, _nbGeneratedNodes + nbGenerated);
    }

    /**
     * Returns a new SearchStatistics instance with an updated current time.
     *
     * @param time the current time (in milliseconds)
     * @return a new instance with the updated time
     */
    public SearchStatistics updateTime(long time) {
        SearchStatistics toReturn = this.copy();
        toReturn._currentTime = time;
        return toReturn;
    }

    /**
     * Creates and returns a copy of this statistics instance with one more node pruned by the fast
     * lower bound.
     *
     * @return a new instance with the incremented number of fast lower bound prunings
     */
    public SearchStatistics incrementNbFastLowerBoundPruning() {
        SearchStatistics toReturn = this.copy();
        toReturn._nbFastLowerBoundPruning++;
        return toReturn;
    }

    /**
     * Creates and returns a copy of this statistics instance with one more node pruned by the
     * dominance.
     *
     * @return a new instance with the incremented number of dominance prunings
     */
    public SearchStatistics incrementNbDominancePruning() {
        SearchStatistics toReturn = this.copy();
        toReturn._nbDominancePruning++;
        return toReturn;
    }

    /**
     * Creates and returns a copy of this statistics instance with additional nodes pruned by the
     * fast lower bound, for instance those pruned during the compilation of a decision diagram.
     *
     * @param nbPruning the number of nodes pruned by the fast lower bound to add
     * @return a new instance with the updated number of fast lower bound prunings
     */
    public SearchStatistics addNbFastLowerBoundPruning(int nbPruning) {
        SearchStatistics toReturn = this.copy();
        toReturn._nbFastLowerBoundPruning += nbPruning;
        return toReturn;
    }

    /**
     * Creates and returns a copy of this statistics instance with additional nodes pruned by the
     * dominance, for instance those pruned during the compilation of a decision diagram.
     *
     * @param nbPruning the number of nodes pruned by the dominance to add
     * @return a new instance with the updated number of dominance prunings
     */
    public SearchStatistics addNbDominancePruning(int nbPruning) {
        SearchStatistics toReturn = this.copy();
        toReturn._nbDominancePruning += nbPruning;
        return toReturn;
    }

    @Override
    public String toString() {
        DecimalFormatSymbols symbols = new DecimalFormatSymbols(Locale.US);
        symbols.setGroupingSeparator(' ');
        DecimalFormat df = new DecimalFormat("#,##0.##########", symbols);
        DecimalFormat gapFormat = new DecimalFormat("#,##0.####", symbols);

        List<String> labels =
                List.of(
                        "Status",
                        "Iterations",
                        "Frontier Max Size",
                        "Runtime",
                        "Incumbent",
                        "Gap",
                        "Best bound",
                        "Expanded nodes",
                        "Generated nodes",
                        "Relative improvement",
                        "Nb FastLowerBound pruning",
                        "Nb Dominance pruning");

        List<String> values =
                List.of(
                        _status.toString(),
                        df.format(_nbIterations),
                        df.format(_frontierMaxSize),
                        PrettyPrint.formatMs(runtime()),
                        Double.isInfinite(_incumbent) ? "∞" : df.format(_incumbent),
                        Double.isInfinite(_gap) ? "∞" : gapFormat.format(_gap) + " %",
                        Double.isInfinite(bestBound())
                                ? (bestBound() > 0 ? "∞" : "-∞")
                                : df.format(bestBound()),
                        df.format(_nbExpandedNodes),
                        df.format(_nbGeneratedNodes),
                        df.format(relativeImprovement()) + " %",
                        df.format(nbFastLowerBoundPruning()),
                        df.format(nbDominancePruning()));

        int labelSize = labels.stream().mapToInt(String::length).max().orElse(1) + 1;
        int valueSize = values.stream().mapToInt(String::length).max().orElse(1) + 1;
        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < labels.size(); i++) {
            sb.append("\t");
            sb.append(("%-" + labelSize + "s ").formatted(labels.get(i) + ":"));
            sb.append(("%" + valueSize + "s").formatted(values.get(i)));
            sb.append("\n");
        }

        return sb.toString();
    }
}
