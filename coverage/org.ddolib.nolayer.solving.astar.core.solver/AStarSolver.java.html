package org.ddolib.nolayer.solving.astar.core.solver;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Iterator;
import java.util.List;
import java.util.Optional;
import java.util.PriorityQueue;
import java.util.function.BiConsumer;
import java.util.function.Predicate;
import org.ddolib.common.solver.stat.AstarStats;
import org.ddolib.common.solver.stat.SearchStatistics;
import org.ddolib.common.solver.stat.SearchStatus;
import org.ddolib.common.solver.stopcriterion.MemoryLimitException;
import org.ddolib.common.util.debug.DebugLevel;
import org.ddolib.common.util.verbosity.VerboseMode;
import org.ddolib.common.util.verbosity.VerbosityLevel;
import org.ddolib.nolayer.modeling.DefaultFastLowerBound;
import org.ddolib.nolayer.modeling.FastLowerBound;
import org.ddolib.nolayer.modeling.Model;
import org.ddolib.nolayer.modeling.NoLayerDominanceChecker;
import org.ddolib.nolayer.modeling.Problem;
import org.ddolib.nolayer.solver.Solution;
import org.ddolib.nolayer.solver.Solver;
import org.ddolib.nolayer.util.debug.DebugUtil;

/**
 * A* solver implementation for no-layer decision-diagram based optimization models.
 *
 * <p>The solver explores subproblems ordered by their {@code f = g + h} value, where {@code g} is
 * the accumulated cost and {@code h} is a fast lower bound on the residual problem. It maintains
 * open/closed sets and updates the incumbent whenever a better complete assignment is discovered.
 *
 * @param <T> the type of states handled by the model
 */
public final class AStarSolver<T> implements Solver {

    private final Problem<T> problem;
    /** Number of nodes whose successors were computed since the beginning of the search. */
    private long nbExpanded = 0;
    /** Number of successors computed (before any pruning) since the beginning of the search. */
    private long nbGenerated = 0;
    private final FastLowerBound<T> lb;
    private final HashMap<T, Double> closed;
    private final HashMap<T, Double> present;
    private final NoLayerDominanceChecker<T> dominance;
    private final PriorityQueue<SubProblem<T>> open =
            new PriorityQueue<>(Comparator.comparingDouble(SubProblem<T>::f));
    private final SubProblem<T> root;
    private final VerboseMode verboseMode;
    private final DebugLevel debugLevel;

    private final boolean defaultLowerBoundValue;
    private AstarStats statistics;
    private double bestUB;
    private Optional<List<Integer>> bestSol;

    /**
     * Creates a new A* solver for the given model.
     *
     * @param model the model to solve
     */
    public AStarSolver(Model<T> model) {
        this.problem = model.problem();
        this.lb = model.lowerBound();
        this.dominance = model.dominance();
        Optional<Solution> init = Solver.initialSolution(model);
        this.bestUB = init.map(Solution::value).orElse(Double.POSITIVE_INFINITY);
        this.bestSol = init.map(Solution::solution);
        this.present = new HashMap<>();
        this.closed = new HashMap<>();
        this.verboseMode = new VerboseMode(model.verbosityLevel(), 500L);
        this.debugLevel = model.debugMode();
        this.root = constructRoot(problem.initialState(), problem.initialValue());
        this.defaultLowerBoundValue = this.lb instanceof DefaultFastLowerBound<T>;
    }

    private AStarSolver(Model<T> model, T state) {
        this.problem = model.problem();
        this.lb = model.lowerBound();
        this.dominance = model.dominance();
        this.bestUB = Double.POSITIVE_INFINITY;
        this.bestSol = Optional.empty();
        this.present = new HashMap<>();
        this.closed = new HashMap<>();
        this.verboseMode = new VerboseMode(VerbosityLevel.SILENT, 500L);
        this.debugLevel = DebugLevel.OFF;
        this.root = constructRoot(state, 0);
        this.defaultLowerBoundValue = this.lb instanceof DefaultFastLowerBound<T>;
    }

    private SubProblem<T> constructRoot(T state, double value) {
        return new SubProblem<>(state, value, lb.fastLowerBound(state), new ArrayList<>());
    }

    /**
     * {@inheritDoc}
     *
     * <p>When the memory runs out, either because the JVM throws an {@link OutOfMemoryError} or
     * because the frontier reaches the size given to {@link
     * org.ddolib.common.solver.stopcriterion.StopCriterion#maxFrontierSize(int)}, the search
     * stops and returns the best solution found so far with the status {@link
     * SearchStatus#MEMOUT}.
     */
    @Override
    public Solution minimize(
            Predicate<SearchStatistics> limit,
            BiConsumer<List<Integer>, SearchStatistics> onSolution) {
        try {
            return search(limit, onSolution);
        } catch (OutOfMemoryError | MemoryLimitException e) {
            // free the memory of the search before building the result
            open.clear();
            present.clear();
            closed.clear();
            statistics =
                    statistics
                            .updateTime(System.currentTimeMillis())
                            .updateStatus(SearchStatus.MEMOUT)
                            .updateNodeCounts(nbExpanded, nbGenerated);
            return new Solution(bestSolution(), statistics);
        }
    }

    /** Runs the search, see {@link #minimize}. */
    private Solution search(
            Predicate<SearchStatistics> limit,
            BiConsumer<List<Integer>, SearchStatistics> onSolution) {
        statistics = new AstarStats(System.currentTimeMillis(), bestUB);
        if (bestSol.isPresent()) { // initial solution provided by the model
            statistics = statistics.updateStatus(SearchStatus.SAT);
        }
        if (dominance != null) {
            dominance.clear();
        }
        nbExpanded = 0;
        nbGenerated = 0;
        open.add(root);
        present.put(root.getState(), root.f());

        while (!open.isEmpty()) {
            statistics =
                    statistics
                            .incrementNbIter()
                            .updateFrontierMaxSize(open.size())
                            .updateTime(System.currentTimeMillis())
                            .updateGap(gap())
                            .updateBestBound(bestBound())
                            .updateNodeCounts(nbExpanded, nbGenerated);

            if (limit.test(statistics)) {
                statistics = statistics.updateNodeCounts(nbExpanded, nbGenerated);
                return new Solution(bestSolution(), statistics);
            }

            SubProblem<T> sub = open.poll();
            T state = sub.getState();

            if (state != null && dominance.updateDominance(state, sub.getValue())) {
                statistics = statistics.incrementNbDominancePruning();
                continue;
            }

            present.remove(state);

            if (closed.containsKey(state)) {
                continue;
            }

            closed.put(state, sub.f());

            if (sub.f() + 1e-10 > bestUB) {
                statistics = statistics.incrementNbFastLowerBoundPruning();
                continue;
            }

            if (problem.isTarget(state)) {
                // target reached
                bestSol = Optional.of(sub.getPath());
                bestUB = sub.getValue();
                break; // Because it's A*, the first target reached is optimal
            } else {
                addChildren(sub, onSolution);
            }
        }

        if (debugLevel != DebugLevel.OFF) {
            checkFlbAdmissibility();
        }

        statistics = statistics.updateTime(System.currentTimeMillis());
        if (bestSol.isPresent()) {
            statistics = statistics.updateStatus(SearchStatus.OPTIMAL).updateIncumbent(bestUB, 0);
        } else {
            statistics = statistics.updateStatus(SearchStatus.UNSAT);
        }

        statistics = statistics.updateNodeCounts(nbExpanded, nbGenerated);
        return new Solution(bestSolution(), statistics);
    }

    private void addChildren(
            SubProblem<T> subProblem, BiConsumer<List<Integer>, SearchStatistics> onSolution) {
        T state = subProblem.getState();
        Iterator<Integer> domain = problem.domain(state);
        nbExpanded++;
        int nbGeneratedChildren = 0;
        int nbSelectedChildren = 0;
        while (domain.hasNext()) {
            int label = domain.next();

            if (debugLevel != DebugLevel.OFF) {
                DebugUtil.checkHashCodeAndEquality(state, label, problem::transition);
            }

            T newState = problem.transition(state, label);
            double cost = problem.transitionCost(state, label);
            nbGeneratedChildren++;
            double g = subProblem.getValue() + cost;
            List<Integer> newPath = new ArrayList<>(subProblem.getPath());
            newPath.add(label);

            double h = lb.fastLowerBound(newState);
            double f = g + h;

            if (f + 1e-10 > bestUB) {
                statistics = statistics.incrementNbFastLowerBoundPruning();
                continue;
            }

            SubProblem<T> newSub = new SubProblem<>(newState, g, h, newPath);

            Double presentValue = present.get(newState);
            if (presentValue != null && presentValue > newSub.f()) {
                open.add(newSub);
                present.put(newState, newSub.f());
                nbSelectedChildren++;
            } else if (presentValue == null) {
                Double closedValue = closed.get(newState);
                if (closedValue != null && closedValue > newSub.f()) {
                    open.add(newSub);
                    closed.remove(newState);
                    present.put(newState, newSub.f());
                    nbSelectedChildren++;
                } else if (closedValue == null) {
                    open.add(newSub);
                    present.put(newState, newSub.f());
                    nbSelectedChildren++;
                }
            }

            if (problem.isTarget(newState) && newSub.getValue() < bestUB) {
                bestSol = Optional.of(newSub.getPath());
                bestUB = newSub.getValue();
                statistics =
                        statistics.updateIncumbent(bestUB, gap())
                                .updateBestBound(bestBound())
                                .updateNodeCounts(nbExpanded, nbGenerated)
                                .updateStatus(SearchStatus.SAT);
                onSolution.accept(newPath, statistics);
                verboseMode.newBest(bestUB);
            }
        }
        nbGenerated += nbGeneratedChildren;
        statistics = statistics.updateValidChildrenPercent(nbGeneratedChildren, nbSelectedChildren);
    }

    /**
     * Returns a lower bound on the value of the solutions reachable through the given node:
     * {@code f = g + h} when the model provides a fast lower bound. Otherwise, the cost of the
     * path {@code g} is a lower bound only if the remaining costs cannot be negative (see {@link
     * Problem#hasNonNegativeCosts()}); if they can, no lower bound is known.
     *
     * @param node an open node
     * @return a lower bound through the node, or {@link Double#NEGATIVE_INFINITY} if unknown
     */
    private double nodeLowerBound(SubProblem<T> node) {
        if (!defaultLowerBoundValue) {
            return node.f();
        }
        return problem.hasNonNegativeCosts() ? node.getValue() : Double.NEGATIVE_INFINITY;
    }

    /**
     * Returns a lower bound on the optimal value: the smallest {@code f = g + h} of the open
     * nodes, capped by the incumbent. When no node is open, the incumbent is optimal.
     *
     * @return a lower bound on the optimal value
     */
    private double bestBound() {
        if (open.isEmpty()) {
            return bestUB;
        }
        return Math.min(bestUB, nodeLowerBound(open.peek()));
    }

    private double gap() {
        if (Double.isInfinite(bestUB)) {
            return Double.POSITIVE_INFINITY;
        } else if (open.isEmpty()) {
            return 0.0;
        } else {
            double globalLB = Math.min(bestUB, nodeLowerBound(open.peek()));
            return 100 * Math.abs(bestUB - globalLB) / Math.abs(bestUB);
        }
    }

    @Override
    public Optional<Double> bestValue() {
        if (bestSol.isPresent()) {
            return Optional.of(bestUB);
        }
        return Optional.empty();
    }

    @Override
    public List<Integer> bestSolution() {
        return bestSol.orElse(List.of());
    }

    private void checkFlbAdmissibility() {
        HashSet<T> toCheck = new HashSet<>(closed.keySet());
        toCheck.addAll(present.keySet());

        Model<T> model =
                new Model<>() {
                    @Override
                    public Problem<T> problem() {
                        return problem;
                    }

                    @Override
                    public FastLowerBound<T> lowerBound() {
                        return lb;
                    }

                    @Override
                    public NoLayerDominanceChecker<T> dominance() {
                        return dominance;
                    }
                };

        DebugUtil.checkFlbAdmissibility(toCheck, model, state -> new AStarSolver<>(model, state));
    }
}
